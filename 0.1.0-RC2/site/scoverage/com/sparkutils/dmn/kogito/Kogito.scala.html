25   def dmnRuntimeFor(dmnFiles: Seq[dmn.DMNFile], dmnConfiguration: dmn.DMNConfiguration): dmn.DMNRuntime = {
26 
27     val resources = dmnFiles.map{ f =>

30     }.toMap

35         .buildConfiguration()

39   }

43   override def providerForType(inputField: dmn.DMNInputField, debug: Boolean, dmnConfiguration: dmn.DMNConfiguration): dmn.DMNContextProvider[_] = {
44     val (path, expr) = try {
45       (KogitoDMNContextPath(inputField.contextPath), inputField.defaultExpr)
46     } catch {
47       case p: ParseException => throw dmn.DMNException(s"${E.CONTEXT_PROVIDER_PARSE} : ${inputField.fieldExpression}", p)
48     }
49 
50     val config = dmnimpl.utils.configMap(dmnConfiguration)
51 
52     inputField.providerType match {
53       case "" => ContextProviderProxy(path, inputField.stillSetWhenNull, expr, config, providedType = None)
54       case "JSON" => KogitoJSONContextProvider(path, inputField.stillSetWhenNull, expr, providedType = Some(StringType))
55       case t if Try(DataType.fromDDL(t)).isSuccess =>
56         val dataType = DataType.fromDDL(t)
57         contextProviderFromDDL(inputField.stillSetWhenNull, path, expr, config, dataType)
58       case _ =>
59         dmnimpl.utils.loadUnaryContextProvider(inputField.providerType, path, expr)
60     }
61 
62   }
63 
64   override def resultProviderForType(resultProviderType: String, debug: Boolean, dmnConfiguration: dmn.DMNConfiguration): dmn.DMNResultProvider =
65     resultProviderType match {
66       case _ if resultProviderType.toUpperCase == "JSON" =>
67         KogitoJSONResultProvider(debug, dmnimpl.utils.configMap(dmnConfiguration))
68       case t if Try(DataType.fromDDL(t)).isSuccess =>
69         val dataType = DataType.fromDDL(t)
70         dataType match {
71           case s: StructType =>
72             KogitoDDLResult(debug = debug, underlyingType = s, dmnimpl.utils.configMap(dmnConfiguration))
73           case _ => throw new dmn.DMNException(s"ResultProvider type $t is not supported, only JSON and Struct is")
74         }
75       case _ =>
76         dmnimpl.utils.loadResultProvider(resultProviderType, debug)
77     }
